# Danh sách dịch thuật các ngôn ngữ còn lại

Bạn hãy copy nội dung của từng ngôn ngữ tương ứng vào file `strings.xml` trong các thư mục `values-<lang_code>`.

## 1. Tiếng Trung (values-zh)
```xml
<resources>
    <string name="app_name" translatable="false">Pion base</string>
    <string name="hd_wallpaper" translatable="false">HD Wallpaper</string>
    <string name="featured">精选</string>
    <string name="top_wallpaper">热门壁纸</string>
    <string name="see_more">查看更多</string>
    <string name="home_tab">首页</string>
    <string name="categories_tab">分类</string>
    <string name="setting">设置</string>
    <string name="update_profile">更新个人资料</string>
    <string name="language">语言</string>
    <string name="dark_theme">深色主题</string>
    <string name="auto_change_wallpaper">自动更换壁纸</string>
    <string name="enable_notifications">启用通知</string>
    <string name="rate_this_app">评价此应用</string>
    <string name="share_with_friend">分享给朋友</string>
    <string name="select_custom_wallpaper">选择自定义壁纸</string>
    <string name="please_select_language">请选择语言</string>
    <string name="close">关闭</string>
    <string name="cancel">取消</string>
    <string name="this_is_demo">这是一个演示对话框</string>
    <string name="pion_tech">Pion Tech</string>
    <string name="website_url" translatable="false">https://piontech.co</string>
    <string name="facebook">Facebook:</string>
    <string name="telegram">Telegram:</string>
    <string name="piontechglobal" translatable="false">@piontechglobal</string>
    <string name="my_favorites">我的收藏</string>
    <string name="no_favorites_yet">暂无收藏</string>
    <string name="ok">确定</string>
    <string name="next">下一步</string>
    <string name="no_results_found">未找到结果</string>
    <string name="set_as_wallpaper">设为壁纸</string>
    <string name="animals">动物</string>
    <string name="arts">艺术</string>
    <string name="nature">自然</string>
    <string name="layout_ads">布局广告</string>
    <string name="english">英语</string>
</resources>
```
